cmake_minimum_required(VERSION 3.20)
project(hosq_performance LANGUAGES CXX)
find_package(benchmark CONFIG REQUIRED)
set(HOSQ_INCLUDE_DIR "" CACHE PATH "Headers of the revision under measurement")
if(NOT EXISTS "${HOSQ_INCLUDE_DIR}/hosq/hosq.hh")
  message(FATAL_ERROR "Set HOSQ_INCLUDE_DIR to the revision's include directory")
endif()
add_executable(hosq_performance workloads.cc)
target_compile_features(hosq_performance PRIVATE cxx_std_23)
target_compile_options(hosq_performance PRIVATE -O3)
target_compile_definitions(hosq_performance PRIVATE NDEBUG HOSQ_USE_FAST_FLOAT=1)
target_include_directories(hosq_performance PRIVATE "${HOSQ_INCLUDE_DIR}")
target_link_libraries(hosq_performance PRIVATE benchmark::benchmark)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
